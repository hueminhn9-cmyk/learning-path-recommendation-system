from functools import wraps

def admin_required(f):
    @wraps(f)
    def decorated_function(*args, **kwargs):
        if 'user_id' not in session:
            flash('Please log in first.', 'error')
            return redirect(url_for('login'))
        if session.get('user_role') != 'admin':
            flash('Access denied. Administrator privileges required.', 'error')
            return redirect(url_for('dashboard'))
        return f(*args, **kwargs)
    return decorated_function

@app.route('/login', methods=['GET', 'POST'])
def login():
    """Login Page Route with MySQL Verification"""
    if request.method == 'POST':
        email = request.form.get('email')
        password = request.form.get('password')
        
        conn = get_db_connection()
        if conn and conn.is_connected():
            try:
                cursor = conn.cursor(dictionary=True)
                cursor.execute("SELECT * FROM users WHERE email = %s AND password = %s", (email, password))
                user = cursor.fetchone()
                cursor.close()
                conn.close()

                if user:
                    session['logged_in'] = True
                    session['user_id'] = user['id']
                    session['user_email'] = user['email']
                    session['user_name'] = user['name']
                    session['user_role'] = user.get('role', 'student')
                    
                    if session['user_role'] == 'admin':
                        flash('🛡️ Admin detected. Redirecting to Management Console...', 'success')
                        return redirect(url_for('admin_dashboard'))
                    return redirect(url_for('dashboard'))
                else:
                    flash('❌ Invalid email or password.', 'error')
            except Error as e:
                print(f"Login error: {e}")
                flash('An error occurred during login.', 'error')
        else:
            flash('Database connection failed.', 'error')
            
    return render_template('index.html')

@app.route('/dashboard')
def dashboard():
    """Student Dashboard showing progress and recommendations"""
    if not session.get('logged_in'):
        return redirect(url_for('login'))
    
    user_id = session.get('user_id')
    # Fetch user data and latest assessment
    conn = get_db_connection()
    user_data = None
    latest_assessment = None
    recommended_courses = []
    stats = {
        'total_assessments': 0,
        'avg_marks': 0,
        'avg_time': 0,
        'latest_level': 'N/A'
    }

    if conn and conn.is_connected():
        try:
            cursor = conn.cursor(dictionary=True)
            # User Info
            cursor.execute("SELECT * FROM users WHERE id = %s", (user_id,))
            user_data = cursor.fetchone()
            
            # Aggregate Stats
            cursor.execute("SELECT COUNT(*) as total, AVG(marks) as avg_m, AVG(time_spent) as avg_t FROM assessments WHERE user_id = %s", (user_id,))
            agg = cursor.fetchone()
            if agg and agg['total'] > 0:
                stats['total_assessments'] = agg['total']
                stats['avg_marks'] = round(agg['avg_m'] or 0, 1)
                stats['avg_time'] = round(agg['avg_t'] or 0, 1)

            # Latest Assessment
            cursor.execute("SELECT * FROM assessments WHERE user_id = %s ORDER BY created_at DESC LIMIT 1", (user_id,))
            latest_assessment = cursor.fetchone()
            if latest_assessment:
                stats['latest_level'] = latest_assessment['predicted_level']
            
            # Recommended courses based on level
            level = user_data.get('current_level', 'Beginner')
            cursor.execute("SELECT * FROM courses WHERE level = %s", (level,))
            recommended_courses = cursor.fetchall()

            cursor.close()
            conn.close()
        except Error as e:
            print(f"Dashboard error: {e}")

    from datetime import datetime
    now = datetime.now().strftime("%B %d, %Y | %H:%M")

    return render_template('dashboard.html', 
                         user=user_data, 
                         assessment=latest_assessment, 
                         stats=stats,
                         now=now,
                         courses=recommended_courses)

@app.route('/predict', methods=['POST'])
def predict():
    """Handles Assessment Form Submission & AI Prediction"""
    if not session.get('logged_in'):
        return redirect(url_for('login'))
    # 1. PREDICT LEVEL USING NEURAL NETWORK
    marks = int(request.form.get('marks', 0))
    time_spent = float(request.form.get('time_spent', 0))
    subject = request.form.get('subject', 'General')
    roll = request.form.get('roll', 'N/A')
    
    # Predict Interest Level automatically (AI-like logic)
    # High if high marks or high study time
    # Low if both are low
    marks_int = int(marks)
    time_float = float(time_spent)
    
    if marks_int >= 80 and time_float >= 4:
        interest = "High"
    elif marks_int < 40 and time_float < 2:
        interest = "Low"
    else:
        interest = "Medium"
        
    # 0. HANDLE FILE UPLOAD
    marksheet_filename = None
    if 'marksheet' in request.files:
        file = request.files['marksheet']
        if file and file.filename != '' and allowed_file(file.filename):
            from werkzeug.utils import secure_filename
            marksheet_filename = secure_filename(f"{roll}_{file.filename}")
            file.save(os.path.join(app.config['UPLOAD_FOLDER'], marksheet_filename))

    # 1. AI PREDICTION
    if model is None:
        flash('❌ AI model not loaded. Cannot process request.', 'error')
        return redirect(url_for('assessment'))
        
    # Prepare input data matching the original Streamlit logic
    marks_val = int(marks)
    time_val = float(time_spent)
    input_data = np.array([[marks_val, interest_to_number(interest), time_val]])
    input_scaled = scaler.transform(input_data)

    # Get prediction
    prediction = model.predict(input_scaled)
    predicted_class = np.argmax(prediction[0])
    confidence = np.max(prediction[0]) * 100

    # Get level string
    level = level_map.get(predicted_class, "Unknown")
    # 2. GENERATE IMPROVEMENT AREAS & LEARNING PATH
    recommendations = {
        "Data Science": {
            "low": ["Mathematics for Data Science (Linear Algebra, Calculus)", "Data Science Fundamentals", "Basics of Statistics"],
            "med": ["Exploratory Data Analysis (EDA) with Pandas", "Data Visualization (Matplotlib, Seaborn)", "Intermediate Statistics & Probability"],
            "high": ["Advanced Feature Engineering", "Statistical Modeling", "Big Data Analytics with Spark"]
        },
        "Machine Learning": {
            "low": ["Basic Algorithms (Linear Regression, KNN)", "Scikit-Learn Fundamentals", "Data Preprocessing Techniques"],
            "med": ["Supervised vs Unsupervised Learning Deep Dive", "Model Evaluation Metrics (Precision, Recall, F1)", "Cross-Validation & Regularization"],
            "high": ["Neural Networks Architecture", "Ensemble Methods (XGBoost, Random Forest)", "MLOps & Model Deployment"]
        },
        "Deep Learning": {
            "low": ["Neural Networks Basics", "TensorFlow/PyTorch Fundamentals", "Activation Functions"],
            "med": ["Convolutional Neural Networks (CNN)", "Recurrent Neural Networks (RNN)", "Backpropagation & Gradient Descent"],
            "high": ["Generative Adversarial Networks (GANs)", "Transformers & Attention Mechanisms", "Reinforcement Learning"]
        },
        "Big Data": {
            "low": ["Introduction to Big Data Concepts", "Distributed Systems Basics", "HDFS Fundamentals"],
            "med": ["MapReduce Algorithms", "Apache Spark DataFrames", "NoSQL Databases (MongoDB, Cassandra)"],
            "high": ["Real-time Stream Processing (Kafka)", "Advanced Spark Optimization", "Data Lake Architecture"]
        },
        "Cloud Computing": {
            "low": ["Cloud Basics & Service Models (IaaS, PaaS, SaaS)", "Virtualization Concepts", "Intro to AWS/Azure/GCP"],
            "med": ["Cloud Storage Solutions", "Serverless Computing", "Containerization (Docker) & Orchestration"],
            "high": ["Multi-Cloud Deployment", "Infrastructure as Code (Terraform)", "Cloud Security & Compliance"]
        },
        "Web Development": {
            "low": ["HTML5 & CSS3 Basics", "JavaScript Fundamentals", "Responsive Web Design"],
            "med": ["Frontend Frameworks (React/Vue/Angular)", "Backend Routing & APIs (Node.js/Django/Flask)", "Database Integration"],
            "high": ["Full-Stack Application Deployment", "Microservices Architecture", "WebAssembly & Performance Optimization"]
        },
        "General": {
            "low": ["Fundamental Programming", "Logical Reasoning", "Basic Computer Science"],
            "med": ["Data Structures", "Web Development Basics", "Object-Oriented Programming"],
            "high": ["Software Architecture", "Advanced Algorithms", "System Design"]
        }
    }

    # Determine specific recommendations
    subj_node = recommendations.get(subject, recommendations["General"])
    if marks < 50:
        improvement_areas = subj_node["low"]
        next_steps = "Focus on foundational concepts. We recommend completing 2-3 introductory courses before retaking the assessment."
    elif marks < 80:
        improvement_areas = subj_node["med"]
        next_steps = "You have a solid base. Now, practice with real-world datasets and focus on intermediate implementation details."
    else:
        improvement_areas = subj_node["high"]
        next_steps = "Great job! You're ready for advanced specializations. Try building a complex project or contributing to open source."

    # 3. SAVE TO MYSQL DATABASE
    conn = get_db_connection()

        try:

                        resources.append(fallback)
                        seen_urls.add(fallback['url'])

            cursor.close()
            conn.close()
        except Exception as e:
            print(f"Resource fetch error: {e}")

    # Render results page with the data
    return render_template('result.html',

                          next_steps=next_steps,
                          resources=resources)

@app.route('/resources')
def resources_page():
    """Learning Resources Library - Dynamic from Database"""
    if not session.get('logged_in'):
        return redirect(url_for('login'))
        
    courses_data = []
    conn = get_db_connection()
    if conn and conn.is_connected():
        try:
            cursor = conn.cursor(dictionary=True)
            cursor.execute("SELECT * FROM courses ORDER BY level, title")
            courses_data = cursor.fetchall()
            cursor.close()
            conn.close()
        except Error as e:
            print(f"Resources fetch error: {e}")
            
    # Group courses by level and subject for better display
    grouped_courses = {}
    seen_urls = set()
    for course in courses_data:
        if course['url'] in seen_urls:
            continue
        seen_urls.add(course['url'])
        
        lvl = course['level']
        subj = course.get('subject', 'General')
        if lvl not in grouped_courses:
            grouped_courses[lvl] = {}
        if subj not in grouped_courses[lvl]:
            grouped_courses[lvl][subj] = []
        grouped_courses[lvl][subj].append(course)
            
    return render_template('resources.html', grouped_courses=grouped_courses)

        agg = cursor.fetchone()
        if agg:
            stats['total'] = agg['total']
            stats['avg'] = round(agg['avg_m'] or 0, 1)
        cursor.close()
        conn.close()

    return render_template('profile.html', user=user_data, stats=stats)

@app.route('/records')
@admin_required
def records():
    """Student Dashboard Route Fetching from MySQL"""
    if not session.get('logged_in'):
        return redirect(url_for('login'))
        
    records_data = []
    conn = get_db_connection()
    if conn and conn.is_connected():
        try:
            cursor = conn.cursor(dictionary=True)
            # Fetch assessments joined with user info
            query = """
                SELECT a.*, u.name, u.roll_number 
                FROM assessments a 
                JOIN users u ON a.user_id = u.id 
                ORDER BY a.created_at DESC
            """
            cursor.execute(query)
            records_data = cursor.fetchall()
            cursor.close()
            conn.close()
        except Error as e:
            print(f"Record fetch error: {e}")
            flash('Error loading records.', 'error')
            
    return render_template('records.html', records=records_data)
